		in p_RlApp.[DEBTOR].[CONTACTS]

	-- Город подразделения
	public function get_city_depart(d ref[DEPART]) return varchar2(128) is
		city_str varchar2(128);
		dep ref [depart];
	begin
		if d.[DEPART_GROUP].[NAME]='Дополнительный офис без кассового узла' then
			return 'Мини ДО';
		end if;
		-- можно попробовать использовать настройку:
		-- if nvl(::[FP_TUNE].[LIB].get_bool_value('POTREB_MINI_DO',::[FP_TUNE].[LIB].set_canons('DEPART',p_rl_app.[DEPART]),'0')	, false) = true then
		-- если "1", то это МиниДО
		-- если "0", то остальные офисы
		-- Город берем из адреса подразделения с кодом 001-ХХ, то есть, с 6-значного кода
		-- Если код длиннее, то нижние по уровню подразделения расположены в поселках (Врангель, Ливадия, Пограничный) и не попадают в выборку;
		-- Точку выдачи берем из наименования подразделения, у которого непустой Тип подразделения; если пустой, то поднимаемся выше по иерархии
		city_str := null;
		dep := d;
		while length(dep.[CODE])> 6
		loop
			dep := dep.[HIGH];
			if dep is null then
				return 'г Владивосток';
			end if;
		end loop;
		while city_str is null
		loop
			for t in dep.[ADDRESSES] where t.[TYPE].[KOD] = 'CORP'
			loop
				city_str := replace(nvl([RUNTIME].[VIEWFUN].GetAddressByRef(t, '$CITY '),[RUNTIME].[VIEWFUN].GetAddress(dep.[FILIAL].[BANK], '$CITY ', 'CORP')),
					'г. Владивосток', 'г Владивосток');
				city_str := replace(city_str, 'г.', 'г');
				return city_str;
			end loop;
			dep := dep.[HIGH];
			if dep is null then
				return 'г Владивосток';
			end if;
		end loop;
		return city_str;
	end;
	-- Получить все города
	public function get_citys_all(v_ct V_CITYS%type) return varchar2 is
		t integer;
		citys_all varchar2(2000) :='#';
	begin
		for t in 2..v_ct.count
		loop
			if v_ct(t).[BOOL] then
				citys_all := citys_all || v_ct(t).[STR] || '#';
			end if;
		end loop;
		citys_all := replace(citys_all, ' ', '');
		return citys_all;
	end;

	function init_temp_sheme_tbl return &temp.Temp_table is
		t_val 			[DNM_TEMP_TBL];
		t_val_ref ref 	[DNM_TEMP_TBL];
		t_tbl 			&temp.Temp_table;
	begin
		&temp.ClearSessObj;
		t_val%init(true, true);
		t_val.[SESSION_ID] := &temp.Get_session_id;
		t_val.[CLASS_ID] := ::[DNMR_SCHEME]%class;
		t_val.[V1] := 'Все';
		t_val.[B1] := true;
		t_val.[N1] := t_tbl.count + 1;
		t_val_ref := ::[DNM_TEMP_TBL]%insert(t_val);
		t_tbl(t_tbl.count + 1):= t_val_ref;
		for u in ::[DNMR_SCHEME] all

		loop
			t_val := null;

			t_val%init(true, true);
			t_val.[SESSION_ID] := &temp.Get_session_id;
			t_val.[CLASS_ID] := ::[DNMR_SCHEME]%class;

			t_tbl(t_tbl.count + 1):= t_val_ref;

	end;
	-- Изменение признкаа выбора схем кредитования
	procedure ChangeChooseStage(t_tbl in out	&temp.Temp_table) is
		idx number;
	begin
		t_tbl(1).[B1] := not t_tbl(1).[B1];
		idx := t_tbl.first;
		while idx  is not null
		loop
			if idx = 1 then
				idx := t_tbl.next(idx);
			continue;
			end if;
			t_tbl(idx).[B1] := t_tbl(1).[B1];
			idx := t_tbl.next(idx);
		end loop;
	end;

	end;
